cmake_minimum_required(VERSION 3.21)
option(USE_HIP "Build for AMD GPUs" OFF)

set(ROCM_PATH "/opt/rocm" CACHE PATH "ROCm install path")
if(USE_HIP)
    list(APPEND CMAKE_PREFIX_PATH "${ROCM_PATH}")
    project(cudisc LANGUAGES CXX HIP)
    find_package(hip       REQUIRED CONFIG)
    find_package(hipblas   REQUIRED CONFIG)
    find_package(hipsparse REQUIRED CONFIG)
    set(GPU_LIBS hip::device roc::hipblas roc::hipsparse)
    set(SRC_DIR     "${HIP_SRC_DIR}/src")
    set(HEADER_DIR  "${HIP_SRC_DIR}/headers")
else()
    project(cudisc LANGUAGES CXX CUDA)
    find_package(CUDAToolkit REQUIRED)
    set(CMAKE_CUDA_ARCHITECTURES 60 70 75 80 86)
    set(GPU_LIBS CUDA::cudart CUDA::cublas CUDA::cusparse)
    set(SRC_DIR    "${CMAKE_SOURCE_DIR}/src")
    set(HEADER_DIR "${CMAKE_SOURCE_DIR}/headers")
endif()

file(GLOB SRC "${SRC_DIR}/*.cu" "${SRC_DIR}/*.cpp")
list(REMOVE_ITEM SRC "${SRC_DIR}/rpsolver.cu")
add_library(cudisc STATIC ${SRC})
if(USE_HIP)
    set_source_files_properties(${SRC} PROPERTIES LANGUAGE HIP)
    set_target_properties(cudisc PROPERTIES LINKER_LANGUAGE HIP)
endif()
target_include_directories(cudisc PUBLIC "${HEADER_DIR}" "${SRC_DIR}" "${ROCM_PATH}/include/hipblas" "${ROCM_PATH}/include/hipsparse")
target_link_libraries(cudisc PUBLIC ${GPU_LIBS})
target_compile_definitions(cudisc PRIVATE OPAC_DIR="${CMAKE_SOURCE_DIR}/codes/opacities")
set_target_properties(cudisc PROPERTIES CXX_STANDARD 17 HIP_STANDARD 17 CUDA_STANDARD 17)

foreach(PATTERN
    "${CMAKE_SOURCE_DIR}/codes/*.cpp"
    "${CMAKE_SOURCE_DIR}/tests/codes/test_*.cpp"
    "${CMAKE_SOURCE_DIR}/tests/codes/test_*.cu"
    "${CMAKE_SOURCE_DIR}/unit_tests/unit_*.cpp")
    file(GLOB MATCHED ${PATTERN})
    foreach(F ${MATCHED})
        get_filename_component(NAME ${F} NAME_WE)
        add_executable(${NAME} ${F})
        if(USE_HIP)
            set_source_files_properties(${F} PROPERTIES LANGUAGE HIP)
            set_target_properties(${NAME} PROPERTIES LINKER_LANGUAGE HIP)
        endif()
        target_link_libraries(${NAME} PRIVATE cudisc)
        target_compile_definitions(${NAME} PRIVATE OPAC_DIR="${CMAKE_SOURCE_DIR}/codes/opacities")
        set_target_properties(${NAME} PROPERTIES CXX_STANDARD 17 HIP_STANDARD 17 CUDA_STANDARD 17)
    endforeach()
endforeach()
